pub fn respond_from(
    zone: &Zone,
    wire: &[u8],
    transport_limit: usize,
    client: IpAddr,

    let resolver = zone
        .has_anames()

        .transpose()?;
    respond_over_transport(
        zone,
        resolver.as_ref(),
        wire,
        transport_limit,
        true,
        Some(client),
    )
}
